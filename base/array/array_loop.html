C++ — Масиви та цикли

1. Перебір елементів масиву циклом for

Оскільки елементи масиву мають послідовні індекси від 0 до N - 1, цикл for ідеально підходить для їхньої обробки.

#include <iostream>
using namespace std;

int main() {
    const int SIZE = 5;
    int numbers[SIZE] = {4, 8, 15, 16, 23};

    // Виведення елементів масиву на екран
    cout << "Елементи масиву: ";
    for (int i = 0; i < SIZE; i++) {
        cout << numbers[i] << " ";
    }
    cout << endl;

    return 0;
}

2. Заповнення масиву даними від користувача

Щоб заповнити масив значеннями з клавіатури, використовується виклик cin всередині циклу:

#include <iostream>
using namespace std;

int main() {
    const int SIZE = 4;
    int data[SIZE];

    cout << "Введіть " << SIZE << " цілих числа:" << endl;
    for (int i = 0; i < SIZE; i++) {
        cout << "data[" << i << "] = ";
        cin >> data[i];
    }

    return 0;
}

3. Базові обчислення над масивами

Часті завдання при роботі з масивами — знаходження суми, середнього арифметичного або пошук елементів.

#include <iostream>
using namespace std;

int main() {
    const int SIZE = 5;
    int arr[SIZE] = {12, 7, 19, 3, 5};

    // Обчислення суми
    int sum = 0;
    for (int i = 0; i < SIZE; i++) {
        sum += arr[i];
    }

    // Пошук максимального елемента
    int maxVal = arr[0];
    for (int i = 1; i < SIZE; i++) {
        if (arr[i] > maxVal) {
            maxVal = arr[i];
        }
    }

    cout << "Сума: " << sum << endl;
    cout << "Максимум: " << maxVal << endl;

    return 0;
}
Порада: Завжди ініціалізуйте змінну для збереження мінімуму/максимуму першим елементом масиву (arr[0]), а не довільним числом на зразок 0.
